\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{tikz-3dplot}

% --- View and lighting setup ----------------------------------------------
\tdplotsetmaincoords{60}{120}

% Camera-look-from direction in world coords for tdplot's convention:
% tdplot rotates world by rotx around x, then rotz around z. Camera is
% at +z' (the doubly-rotated +z), which in world coords is:
%   ( sin(rotx)*sin(rotz), -sin(rotx)*cos(rotz), cos(rotx) )
\pgfmathsetmacro{\viewX}{sin(60)*sin(120)}
\pgfmathsetmacro{\viewY}{-sin(60)*cos(120)}
\pgfmathsetmacro{\viewZ}{cos(60)}

\pgfmathsetmacro{\lightX}{1/sqrt(6)}
\pgfmathsetmacro{\lightY}{1/sqrt(6)}
\pgfmathsetmacro{\lightZ}{2/sqrt(6)}

% --- Geometry constants ---------------------------------------------------
\pgfmathsetmacro{\PHI}{(1+sqrt(5))/2}
\pgfmathsetmacro{\IPHI}{(sqrt(5)-1)/2}
\pgfmathsetmacro{\TET}{1/sqrt(3)}
\pgfmathsetmacro{\ICOS}{1/sqrt(2+\PHI)}
\pgfmathsetmacro{\PHIICOS}{\PHI*\ICOS}
\pgfmathsetmacro{\PHITET}{\PHI*\TET}
\pgfmathsetmacro{\IPHITET}{\IPHI*\TET}

% --- Vertex storage -------------------------------------------------------
\newcommand{\PVert}[5]{%
  \expandafter\edef\csname Vx#1at#2\endcsname{#3}%
  \expandafter\edef\csname Vy#1at#2\endcsname{#4}%
  \expandafter\edef\csname Vz#1at#2\endcsname{#5}%
}
\newcommand{\Vx}[2]{\csname Vx#1at#2\endcsname}
\newcommand{\Vy}[2]{\csname Vy#1at#2\endcsname}
\newcommand{\Vz}[2]{\csname Vz#1at#2\endcsname}

% --- Per-face shading (back-face cull + Lambertian) -----------------------
% \faceShade{name}{i}{j}{k} sets \NdotV (visibility test, draw iff > 0)
% and \graypct (fill black!\graypct, in [10,35]). Auto-orients the normal
% by checking against the face centroid, so face vertex order need not be
% CCW-from-outside.
\newcommand{\faceShade}[4]{%
  \pgfmathsetmacro{\Pax}{\Vx{#1}{#2}}\pgfmathsetmacro{\Pay}{\Vy{#1}{#2}}\pgfmathsetmacro{\Paz}{\Vz{#1}{#2}}%
  \pgfmathsetmacro{\Pbx}{\Vx{#1}{#3}}\pgfmathsetmacro{\Pby}{\Vy{#1}{#3}}\pgfmathsetmacro{\Pbz}{\Vz{#1}{#3}}%
  \pgfmathsetmacro{\Pcx}{\Vx{#1}{#4}}\pgfmathsetmacro{\Pcy}{\Vy{#1}{#4}}\pgfmathsetmacro{\Pcz}{\Vz{#1}{#4}}%
  \pgfmathsetmacro{\Eax}{\Pbx-\Pax}\pgfmathsetmacro{\Eay}{\Pby-\Pay}\pgfmathsetmacro{\Eaz}{\Pbz-\Paz}%
  \pgfmathsetmacro{\Ebx}{\Pcx-\Pax}\pgfmathsetmacro{\Eby}{\Pcy-\Pay}\pgfmathsetmacro{\Ebz}{\Pcz-\Paz}%
  \pgfmathsetmacro{\Nrx}{\Eay*\Ebz - \Eaz*\Eby}%
  \pgfmathsetmacro{\Nry}{\Eaz*\Ebx - \Eax*\Ebz}%
  \pgfmathsetmacro{\Nrz}{\Eax*\Eby - \Eay*\Ebx}%
  \pgfmathsetmacro{\Nlen}{sqrt(\Nrx*\Nrx + \Nry*\Nry + \Nrz*\Nrz)}%
  \pgfmathsetmacro{\Nx}{\Nrx/\Nlen}%
  \pgfmathsetmacro{\Ny}{\Nry/\Nlen}%
  \pgfmathsetmacro{\Nz}{\Nrz/\Nlen}%
  \pgfmathsetmacro{\Cx}{(\Pax+\Pbx+\Pcx)/3}%
  \pgfmathsetmacro{\Cy}{(\Pay+\Pby+\Pcy)/3}%
  \pgfmathsetmacro{\Cz}{(\Paz+\Pbz+\Pcz)/3}%
  \pgfmathsetmacro{\NdotC}{\Nx*\Cx + \Ny*\Cy + \Nz*\Cz}%
  \pgfmathparse{\NdotC<0 ? 1 : 0}%
  \ifnum\pgfmathresult=1
    \pgfmathsetmacro{\Nx}{-1*\Nx}\pgfmathsetmacro{\Ny}{-1*\Ny}\pgfmathsetmacro{\Nz}{-1*\Nz}%
  \fi
  \pgfmathsetmacro{\NdotV}{\Nx*\viewX + \Ny*\viewY + \Nz*\viewZ}%
  \pgfmathsetmacro{\NdotL}{max(0, \Nx*\lightX + \Ny*\lightY + \Nz*\lightZ)}%
  \pgfmathsetmacro{\graypct}{int(round(35 - 25*\NdotL))}%
}

% --- Fixed-arity face drawers --------------------------------------------
\newcommand{\drawTri}[4]{%
  \faceShade{#1}{#2}{#3}{#4}%
  \pgfmathparse{\NdotV>0 ? 1 : 0}%
  \ifnum\pgfmathresult=1
    \draw[fill=black!\graypct, draw=black!75, line width=0.4pt, line join=round]
      (\Vx{#1}{#2},\Vy{#1}{#2},\Vz{#1}{#2}) --
      (\Vx{#1}{#3},\Vy{#1}{#3},\Vz{#1}{#3}) --
      (\Vx{#1}{#4},\Vy{#1}{#4},\Vz{#1}{#4}) -- cycle;
  \fi
}
\newcommand{\drawQuad}[5]{%
  \faceShade{#1}{#2}{#3}{#4}%
  \pgfmathparse{\NdotV>0 ? 1 : 0}%
  \ifnum\pgfmathresult=1
    \draw[fill=black!\graypct, draw=black!75, line width=0.4pt, line join=round]
      (\Vx{#1}{#2},\Vy{#1}{#2},\Vz{#1}{#2}) --
      (\Vx{#1}{#3},\Vy{#1}{#3},\Vz{#1}{#3}) --
      (\Vx{#1}{#4},\Vy{#1}{#4},\Vz{#1}{#4}) --
      (\Vx{#1}{#5},\Vy{#1}{#5},\Vz{#1}{#5}) -- cycle;
  \fi
}
\newcommand{\drawPent}[6]{%
  \faceShade{#1}{#2}{#3}{#4}%
  \pgfmathparse{\NdotV>0 ? 1 : 0}%
  \ifnum\pgfmathresult=1
    \draw[fill=black!\graypct, draw=black!75, line width=0.4pt, line join=round]
      (\Vx{#1}{#2},\Vy{#1}{#2},\Vz{#1}{#2}) --
      (\Vx{#1}{#3},\Vy{#1}{#3},\Vz{#1}{#3}) --
      (\Vx{#1}{#4},\Vy{#1}{#4},\Vz{#1}{#4}) --
      (\Vx{#1}{#5},\Vy{#1}{#5},\Vz{#1}{#5}) --
      (\Vx{#1}{#6},\Vy{#1}{#6},\Vz{#1}{#6}) -- cycle;
  \fi
}
